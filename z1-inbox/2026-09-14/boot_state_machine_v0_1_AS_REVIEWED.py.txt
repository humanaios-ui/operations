@dataclass
class TransitionResult:
    success: bool
    new_state: BootState
    reason: str
    details: Dict[str, Any] = field(default_factory=dict)


@dataclass
class MachineContext:
    """Mutable session context the machine and agent share."""
    registry_touching: bool = False
    live_state_ok: bool = False
    path_c: bool = False                # Slack MCP unavailable → PATH_C
    commit_pin_ok: bool = False
    registered_fetch_ok: bool = False
    registered_class_state: str = "UNKNOWN"  # OK | UNAVAILABLE | UNKNOWN | STALE
    confirmation_received: bool = False
    b0_verification_ok: bool = False
    work_done: bool = False
    close_artifact_emitted: bool = False
    notes: List[str] = field(default_factory=list)


class BootStateMachine:
    """
    Minimal advisory state machine.
    Agent actions are validated against current state; transitions are
    driven by explicit guards that restate existing protocol language.
    """

    def __init__(self, registry_touching: bool = False):
        self.state = BootState.BS_POWER_ON
        self.ctx = MachineContext(registry_touching=registry_touching)
        self.history: List[str] = []
        self._log(f"INIT registry_touching={registry_touching}")

    def _log(self, msg: str) -> None:
        entry = f"[{self.state.name}] {msg}"
        self.history.append(entry)
        print(entry)

    def current_state(self) -> BootState:
        return self.state

    def allowed_actions(self) -> List[str]:
        return ALLOWED_ACTIONS.get(self.state, [])

    def can_act(self, action: str) -> bool:
        return action in self.allowed_actions()

    # ------------------------------------------------------------------
    # Guards / transition logic (faithful to existing language)
    # ------------------------------------------------------------------

    def transition(self, action: str, **kwargs) -> TransitionResult:
        if not self.can_act(action):
            return TransitionResult(
                False, self.state,
                f"Action '{action}' not allowed in state {self.state.name}"
            )

        handler = getattr(self, f"_do_{action}", None)
        if handler is None:
            return TransitionResult(False, self.state, f"No handler for action '{action}'")

        result = handler(**kwargs)
        if result.success and result.new_state != self.state:
            old = self.state
            self.state = result.new_state
            self._log(f"TRANSITION {old.name} → {self.state.name} | {result.reason}")
        else:
            self._log(f"ACTION {action} | {result.reason}")
        return result

    # --- action handlers ---

    def _do_start(self) -> TransitionResult:
        return TransitionResult(True, BootState.BS_POST, "Power-on → POST")

    def _do_fetch_live_state(self, slack_mcp_available: bool = True) -> TransitionResult:
        if not slack_mcp_available:
            self.ctx.path_c = True
            self.ctx.live_state_ok = False
            self.ctx.notes.append("PATH_C declared (Slack MCP unavailable)")
            # OPERATOR_RUNBOOK: declare PATH_C and proceed from CURRENT.md only
            return TransitionResult(
                True, BootState.BS_SECURE_BOOT,
                "PATH_C (degraded live-state path) — proceed with CURRENT.md only"
            )
        self.ctx.live_state_ok = True
        return TransitionResult(True, BootState.BS_SECURE_BOOT, "Live state OK")

    def _do_pin_commit(self, pin_ok: bool = True) -> TransitionResult:
        self.ctx.commit_pin_ok = pin_ok
        if not pin_ok:
            # §F.1 style: stop and ask (softened from pure terminal halt)
            return TransitionResult(
                True, BootState.BS_HALTED,
                "Commit pin failed — stop and ask operator (§F.1)"
            )
        return TransitionResult(True, BootState.BS_BOOTLOADER, "Commit pin OK")
